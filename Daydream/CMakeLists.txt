project(DAYDREAM)

file(GLOB_RECURSE DAYDREAM_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.h ${CMAKE_CURRENT_SOURCE_DIR}/*.hpp)
file(GLOB_RECURSE DAYDREAM_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)

add_executable(DAYDREAM ${DAYDREAM_HEADERS} ${DAYDREAM_SOURCES})

target_precompile_headers(DAYDREAM PRIVATE ../Imagination/src/pch.hpp)
target_include_directories(DAYDREAM PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../Imagination/src)

target_link_libraries(DAYDREAM PRIVATE IMGN)

target_compile_definitions(DAYDREAM PRIVATE IMGN_PLATFORM_WINDOWS)

set_target_properties(DAYDREAM PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${OUTPUT_DIR}/Daydream LIBRARY_OUTPUT_DIRECTORY ${OUTPUT_DIR}/Daydream ARCHIVE_OUTPUT_DIRECTORY ${OUTPUT_DIR}/Daydream OBJECT_OUTPUTS_DIRECTORY ${OBJ_DIR}/Daydream)
